=== docs/decisions.md ===
# Decisions

- **No separate users table; Supabase `auth.users` is the user record.** Reason: a profiles table would copy the email (same fact in two places). Rejected: `public.users` mirror table.
- **School domain is checked with an exact match on the part after `@`, at signup and again on every API request.** Reason: blocks `evil-school.ac.kr` / `school.ac.kr.evil.com` bypasses and accounts whose email changed. Rejected: `endsWith(domain)` check.
- **Domain lives only in env `SCHOOL_EMAIL_DOMAIN`; RLS checks "email confirmed" + ownership, the server layer checks the domain.** Reason: one source for the domain. Rejected: copying the domain into a DB settings table for RLS.
- **Sale status is derived from `sold_at` (null = 판매중).** Reason: no stored status column that can disagree with `sold_at`. Rejected: `status` enum + `sold_at`.
- **Sold posts stay on the main list and My Page for 30 days after `sold_at` (labelled 판매 완료), then are excluded from both.** Reason: rule "30일간 마이페이지에서 조회 … 이후 메인 목록에서 제외". Rows are not deleted; filtering happens at query time. Rejected: a cron job that deletes/flags rows.
- **One offer per buyer per post (DB unique constraint), duplicate returns 409.** Reason: prevents duplicate offers even under concurrent requests. Rejected: app-only duplicate check.
- **Photos go to a private Supabase Storage bucket and are shown through short-lived signed URLs.** Reason: no public write/list; file type is checked by magic bytes on the server, not by the client's MIME header. Rejected: public bucket.
- **Rate limiting is in-memory per server instance.** Reason: zero-cost; known limit: on Vercel each instance has its own counter. Rejected for now: Redis/Upstash.

=== lib/rules/email.ts ===
export function emailDomain(email: string): string | null {
  const m = /^[^\s@]+@([a-z0-9.-]+)$/.exec(email.trim().toLowerCase());
  return m ? m[1] : null;
}

/** Exact domain match only: subdomains and look-alike suffixes are rejected. */
export function isSchoolEmail(email: string, schoolDomain: string): boolean {
  const domain = emailDomain(email);
  return domain !== null && schoolDomain !== "" && domain === schoolDomain.trim().toLowerCase();
}

=== lib/rules/validate.ts ===
export type Result<T> = { ok: true; value: T } | { ok: false; error: string };

export function text(raw: unknown, label: string, min: number, max: number): Result<string> {
  if (typeof raw !== "string") return { ok: false, error: `${label}을(를) 입력해 주세요.` };
  const v = raw.trim();
  if (v.length < min || v.length > max) return { ok: false, error: `${label}은(는) ${min}~${max}자여야 합니다.` };
  return { ok: true, value: v };
}

export function wholeNumber(raw: unknown, label: string, min: number, max: number): Result<number> {
  const s = typeof raw === "number" ? String(raw) : typeof raw === "string" ? raw.trim() : "";
  if (!/^\d+$/.test(s)) return { ok: false, error: `${label}은(는) 숫자로만 입력해 주세요.` };
  const n = Number(s);
  if (n < min || n > max) return { ok: false, error: `${label}은(는) ${min}~${max} 사이여야 합니다.` };
  return { ok: true, value: n };
}

import { Result, text, wholeNumber } from "./validate";

export const SOLD_RETENTION_DAYS = 30;
export const MAX_PRICE = 10_000_000;

export type ListingInput = { title: string; description: string; price: number; allowOffers: boolean };

export function parseListing(raw: Record<string, unknown>): Result<ListingInput> {
  const title = text(raw.title, "제목", 1, 100);
  if (!title.ok) return title;
  const description = text(raw.description ?? "", "설명", 0, 2000);
  if (!description.ok) return description;
  const price = wholeNumber(raw.price, "판매 가격", 0, MAX_PRICE); // price is mandatory
  if (!price.ok) return price;
  if (raw.allowOffers !== "true" && raw.allowOffers !== "false")
    return { ok: false, error: "가격 제안 허용 여부를 선택해 주세요." };
  return { ok: true, value: { title: title.value, description: description.value, price: price.value, allowOffers: raw.allowOffers === "true" } };
}

/** Sold posts older than this instant are hidden from the main list and My Page. */
export function retentionCutoff(now = new Date()): Date {
  return new Date(now.getTime() - SOLD_RETENTION_DAYS * 24 * 60 * 60 * 1000);
}

export function isVisible(soldAt: string | null, now = new Date()): boolean {
  return soldAt === null || new Date(soldAt) >= retentionCutoff(now);
}

export function statusLabel(soldAt: string | null): string {
  return soldAt === null ? "판매중" : "판매 완료";
}

=== lib/rules/upload.ts ===
export const MAX_PHOTO_BYTES = 5 * 1024 * 1024;

const SIGNATURES: { ext: string; mime: string; test: (b: Uint8Array) => boolean }[] = [
  { ext: "jpg", mime: "image/jpeg", test: (b) => b[0] === 0xff && b[1] === 0xd8 && b[2] === 0xff },
  { ext: "png", mime: "image/png", test: (b) => b[0] === 0x89 && b[1] === 0x50 && b[2] === 0x4e && b[3] === 0x47 },
  { ext: "webp", mime: "image/webp", test: (b) => String.fromCharCode(...b.slice(0, 4)) === "RIFF" && String.fromCharCode(...b.slice(8, 12)) === "WEBP" },
];

/** Detects the type from file bytes, ignoring the client-supplied MIME type. */
export function detectImage(bytes: Uint8Array): { ext: string; mime: string } | null {
  const hit = SIGNATURES.find((s) => s.test(bytes));
  return hit ? { ext: hit.ext, mime: hit.mime } : null;
}

export function handle<A extends unknown[]>(fn: (...args: A) => Promise<Response>) {
  return async (...args: A): Promise<Response> => {
    try {
      return await fn(...args);
    } catch (e) {
      if (e instanceof HttpError) return Response.json({ error: e.message }, { status: e.status });
      console.error(e);
      return Response.json({ error: "서버 오류가 발생했습니다." }, { status: 500 });
    }
  };
}

=== lib/server/rateLimit.ts ===
const hits = new Map<string, number[]>();

/** In-memory sliding window; see docs/decisions.md for the multi-instance limit. */
export function rateLimit(key: string, max: number, windowMs: number): boolean {
  const now = Date.now();
  const recent = (hits.get(key) ?? []).filter((t) => now - t < windowMs);
  if (recent.length >= max) { hits.set(key, recent); return false; }
  recent.push(now);
  hits.set(key, recent);
  return true;
}

import { SupabaseClient } from "@supabase/supabase-js";
import { retentionCutoff, statusLabel } from "../rules/listing";

export const POST_COLUMNS = "id, seller_id, title, description, price, allow_offers, photo_path, created_at, sold_at";
export type PostRow = {
  id: string; seller_id: string; title: string; description: string; price: number;
  allow_offers: boolean; photo_path: string; created_at: string; sold_at: string | null;
};
export const PHOTO_BUCKET = "book-photos";

export async function listVisiblePosts(sb: SupabaseClient, sellerId?: string): Promise<PostRow[]> {
  let q = sb.from("posts").select(POST_COLUMNS)
    .or(`sold_at.is.null,sold_at.gte.${retentionCutoff().toISOString()}`)
    .order("created_at", { ascending: false }).limit(100);
  if (sellerId) q = q.eq("seller_id", sellerId);
  const { data, error } = await q;
  if (error) throw error;
  return data as PostRow[];
}

export async function toViews(sb: SupabaseClient, rows: PostRow[], userId: string) {
  if (rows.length === 0) return [];
  const { data, error } = await sb.storage.from(PHOTO_BUCKET).createSignedUrls(rows.map((r) => r.photo_path), 600);
  if (error) throw error;
  return rows.map((r, i) => ({
    id: r.id, title: r.title, description: r.description, price: r.price, allowOffers: r.allow_offers,
    createdAt: r.created_at, soldAt: r.sold_at, status: statusLabel(r.sold_at),
    isMine: r.seller_id === userId, photoUrl: data[i]?.signedUrl ?? null,
  }));
}

import { randomUUID } from "crypto";
import { parseListing } from "@/lib/rules/listing";
import { detectImage, MAX_PHOTO_BYTES } from "@/lib/rules/upload";
import { handle, HttpError } from "@/lib/server/http";
import { listVisiblePosts, PHOTO_BUCKET, toViews } from "@/lib/server/posts";
import { requireMember } from "@/lib/server/session";

export const GET = handle(async () => {
  const { supabase, user } = await requireMember();
  return Response.json(await toViews(supabase, await listVisiblePosts(supabase), user.id));
});

export const POST = handle(async (req: Request) => {
  const { supabase, user } = await requireMember();
  const form = await req.formData();
  const listing = parseListing(Object.fromEntries([...form.entries()].filter(([, v]) => typeof v === "string")));
  if (!listing.ok) throw new HttpError(400, listing.error);
  const photo = form.get("photo");
  if (!(photo instanceof File) || photo.size === 0) throw new HttpError(400, "책 사진을 첨부해 주세요.");
  if (photo.size > MAX_PHOTO_BYTES) throw new HttpError(413, "사진은 5MB 이하만 올릴 수 있습니다.");
  const bytes = new Uint8Array(await photo.arrayBuffer());
  const type = detectImage(bytes);
  if (!type) throw new HttpError(415, "JPG, PNG, WEBP 사진만 올릴 수 있습니다.");
  const path = `${user.id}/${randomUUID()}.${type.ext}`;
  const up = await supabase.storage.from(PHOTO_BUCKET).upload(path, bytes, { contentType: type.mime });
  if (up.error) throw up.error;
  const { title, description, price, allowOffers } = listing.value;
  const { data, error } = await supabase.from("posts")
    .insert({ seller_id: user.id, title, description, price, allow_offers: allowOffers, photo_path: path })
    .select("id").single();
  if (error) {
    const rm = await supabase.storage.from(PHOTO_BUCKET).remove([path]);
    if (rm.error) console.error("orphan photo cleanup failed", path, rm.error);
    throw error;
  }
  return Response.json({ id: data.id }, { status: 201 });
});

export async function api<T>(url: string, init?: RequestInit): Promise<T> {
  const res = await fetch(url, init);
  const body = await res.json().catch(() => ({ error: `요청 실패 (${res.status})` }));
  if (!res.ok) throw new Error(body.error ?? `요청 실패 (${res.status})`);
  return body as T;
}

"use client";
import Link from "next/link";
import { useEffect, useState } from "react";
import { api, PostView } from "@/lib/client/api";

export function PostList({ source }: { source: string }) {
  const [posts, setPosts] = useState<PostView[] | null>(null);
  const [error, setError] = useState("");
  useEffect(() => { api<PostView[]>(source).then(setPosts).catch((e) => setError(e.message)); }, [source]);
  if (error) return <p role="alert">{error}</p>;
  if (!posts) return <p>불러오는 중…</p>;
  if (posts.length === 0) return <p>게시글이 없습니다.</p>;
  return (
    <ul>
      {posts.map((p) => (
        <li key={p.id}>
          <Link href={`/posts/${p.id}`}>{p.title}</Link> · {p.price.toLocaleString()}원 · {p.status}
          {p.allowOffers && !p.soldAt ? " · 가격 제안 가능" : ""}
        </li>
      ))}
    </ul>
  );
}

"use client";
import { FormEvent, useState } from "react";
import { api } from "@/lib/client/api";

export function AuthForm({ endpoint, submitLabel, onDone }: { endpoint: string; submitLabel: string; onDone: (msg?: string) => void }) {
  const [msg, setMsg] = useState("");
  async function submit(e: FormEvent<HTMLFormElement>) {
    e.preventDefault();
    const f = new FormData(e.currentTarget);
    try {
      const r = await api<{ message?: string }>(endpoint, {
        method: "POST", headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ email: f.get("email"), password: f.get("password") }),
      });
      onDone(r.message);
    } catch (err) { setMsg((err as Error).message); }
  }
  return (
    <form onSubmit={submit}>
      <input name="email" type="email" placeholder="학교 이메일" required />
      <input name="password" type="password" placeholder="비밀번호 (8자 이상)" required />
      <button>{submitLabel}</button>
      {msg && <p role="alert">{msg}</p>}
    </form>
  );
}

"use client";
import { FormEvent, use, useCallback, useEffect, useState } from "react";
import { api, PostView } from "@/lib/client/api";

type Offer = { id: string; amount: number; created_at: string };

export default function PostDetail({ params }: { params: Promise<{ id: string }> }) {
  const { id } = use(params);
  const [post, setPost] = useState<PostView | null>(null);
  const [offers, setOffers] = useState<Offer[]>([]);
  const [msg, setMsg] = useState("");
  const fail = (e: Error) => setMsg(e.message);

  const load = useCallback(async () => {
    const p = await api<PostView>(`/api/posts/${id}`);
    setPost(p);
    if (p.isMine) setOffers(await api<Offer[]>(`/api/posts/${id}/offers`));
  }, [id]);
  useEffect(() => { load().catch(fail); }, [load]);

  async function offer(e: FormEvent<HTMLFormElement>) {
    e.preventDefault();
    const amount = new FormData(e.currentTarget).get("amount");
    await api(`/api/posts/${id}/offers`, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ amount }) })
      .then(() => setMsg("가격을 제안했습니다.")).catch(fail);
  }
  async function markSold() {
    await api(`/api/posts/${id}/sold`, { method: "POST" }).then(load).catch(fail);
  }

  if (!post) return <p role="alert">{msg || "불러오는 중…"}</p>;
  return (
    <article>
      <h1>{post.title}</h1>
      {post.photoUrl && <img src={post.photoUrl} alt={post.title} width={320} />}
      <p>{post.price.toLocaleString()}원 · {post.status}</p>
      <p style={{ whiteSpace: "pre-wrap" }}>{post.description}</p>
      {!post.isMine && post.allowOffers && !post.soldAt && (
        <form onSubmit={offer}>
          <input name="amount" type="number" min={1} step={1} placeholder="제안 가격(원)" required />
          <button>가격 제안</button>
        </form>
      )}
      {post.isMine && (
        <section>
          <h2>받은 가격 제안</h2>
          {offers.length === 0 ? <p>아직 없습니다.</p> :
            <ul>{offers.map((o) => <li key={o.id}>{o.amount.toLocaleString()}원</li>)}</ul>}
          {!post.soldAt && <button onClick={markSold}>판매 완료로 변경</button>}
        </section>
      )}
      {msg && <p role="alert">{msg}</p>}
    </article>
  );
}
