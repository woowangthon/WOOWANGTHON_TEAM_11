=== docs/decisions.md ===
# Design decisions

1. **All writes go through Next.js route handlers using the service-role key; RLS allows members read-only access.**
   Reason: every write is authorized and validated on the server, and the database denies direct writes from browsers even if the API is bypassed.
   Rejected: RLS insert/update policies for clients (fine rules would then live in SQL policies and a member could insert rows directly).
2. **Fines are an append-only ledger (`fine_entries`); balances are computed, never stored.**
   Reason: security rule "never overwrite an amount". Waive = entry bringing the attendance balance to 0; adjust = entry with the difference; payment = positive `payment` entry.
   Rejected: a `fine_amount` column on `attendance` updated in place.
3. **Multi-row writes (attendance + charge + audit) run in one Postgres function, executable only by service_role.** Fine amounts are computed in `lib/domain/rules.ts` and passed in; the function only stores them atomically and checks the expected balance (row lock) so concurrent edits cannot double-apply.
   Rejected: separate supabase-js inserts (partial writes on failure).
4. **The host role is stored once, in `group_members.role`, and created in the same transaction as the group.** `groups.owner_id` is unique to enforce "one group created per user".
5. **Status is decided by the server clock:** check-in at or before `starts_at + grace_minutes` is present, later is late. Members not checked in when the host closes the session are absent. Check-in opens 30 minutes before start. Times are stored as `timestamptz` (UTC) and shown in Asia/Seoul.
6. **Changing an attendance status resets that attendance's fine to the fixed fee of the new status** (a previous manual adjustment for that attendance is superseded; both remain in the ledger).
7. **Payments are recorded by the host (cash/transfer).** No payment provider is integrated, so there are no webhooks; a payment cannot exceed the outstanding balance.
8. **Fees are fixed at group creation and not editable**, so past charges never depend on a changed setting. Rejected: editable fee columns (would be an in-place amount change).
9. **Rate limiting is in memory per server instance** (login start, join-by-code). Rejected for now: Redis/Upstash (extra paid/hosted dependency). On Vercel this limits per instance only.
10. **Dashboard refreshes after each action (router.refresh) rather than Supabase Realtime.** Aggregation is done in `summarize()` over the group's ledger, which is indexed by `(group_id, user_id)`.

-- Atomic writers. Amounts are computed by the app (lib/domain/rules.ts); these only store them.
create function create_group(p_owner uuid, p_name text, p_code text, p_late int, p_absent int, p_grace int)
returns uuid language plpgsql as $$
declare gid uuid;
begin
  insert into groups(owner_id, name, join_code, late_fee, absent_fee, grace_minutes)
  values (p_owner, p_name, p_code, p_late, p_absent, p_grace) returning id into gid;
  insert into group_members(group_id, user_id, role) values (gid, p_owner, 'host');
  insert into audit_log(group_id, actor_id, action, detail)
  values (gid, p_owner, 'group.create', jsonb_build_object('name', p_name, 'late_fee', p_late, 'absent_fee', p_absent));
  return gid;
end $$;

create function record_attendance(p_session uuid, p_user uuid, p_status text, p_charge int, p_actor uuid)
returns void language plpgsql as $$
declare gid uuid; aid uuid;
begin
  select group_id into strict gid from study_sessions where id = p_session;
  insert into attendance(session_id, user_id, status) values (p_session, p_user, p_status) returning id into aid;
  if p_charge <> 0 then
    insert into fine_entries(group_id, user_id, attendance_id, kind, amount, created_by)
    values (gid, p_user, aid, 'charge', p_charge, p_actor);
  end if;
  insert into audit_log(group_id, actor_id, action, detail)
  values (gid, p_actor, 'attendance.record', jsonb_build_object('attendance', aid, 'user', p_user, 'status', p_status, 'charge', p_charge));
end $$;

create function add_fine_entry(p_group uuid, p_user uuid, p_attendance uuid, p_kind text, p_amount int,
  p_reason text, p_actor uuid, p_expected int, p_new_status text default null)
returns void language plpgsql as $$
declare cur int; old_status text;
begin
  if p_attendance is not null then
    select status into strict old_status from attendance where id = p_attendance for update;
    select coalesce(sum(amount), 0) into cur from fine_entries where attendance_id = p_attendance;
    if cur <> p_expected then raise exception 'balance changed, retry' using errcode = '40001'; end if;
    if p_new_status is not null then update attendance set status = p_new_status where id = p_attendance; end if;
  end if;
  if p_amount <> 0 then
    insert into fine_entries(group_id, user_id, attendance_id, kind, amount, reason, created_by)
    values (p_group, p_user, p_attendance, p_kind, p_amount, p_reason, p_actor);
  end if;
  insert into audit_log(group_id, actor_id, action, detail)
  values (p_group, p_actor, 'fine.' || p_kind, jsonb_build_object('user', p_user, 'attendance', p_attendance,
    'amount', p_amount, 'reason', p_reason, 'old_status', old_status, 'new_status', p_new_status));
end $$;

revoke execute on function create_group, record_attendance, add_fine_entry from public, anon, authenticated;
grant execute on function create_group, record_attendance, add_fine_entry to service_role;

// Business rules only: no UI, no network.
export type Status = 'present' | 'late' | 'absent';
export type EntryKind = 'charge' | 'adjust' | 'waive' | 'payment';
export interface FeePolicy { lateFee: number; absentFee: number; graceMinutes: number }
export interface LedgerEntry { userId: string; attendanceId: string | null; kind: EntryKind; amount: number }
export interface MemberSummary { userId: string; fined: number; paid: number; outstanding: number; paidUp: boolean }

export const CHECK_IN_OPENS_BEFORE_MIN = 30;

export class RuleError extends Error {}

/** Fixed-amount rule: late -> late fee, absent -> absent fee, present -> 0. */
export function fineFor(status: Status, policy: FeePolicy): number {
  if (status === 'late') return policy.lateFee;
  if (status === 'absent') return policy.absentFee;
  return 0;
}

export function classifyCheckIn(now: Date, startsAt: Date, closedAt: Date | null, graceMinutes: number): Status {
  if (closedAt) throw new RuleError('이미 마감된 출석입니다');
  if (now.getTime() < startsAt.getTime() - CHECK_IN_OPENS_BEFORE_MIN * 60_000) {
    throw new RuleError(`출석은 시작 ${CHECK_IN_OPENS_BEFORE_MIN}분 전부터 가능합니다`);
  }
  return now.getTime() <= startsAt.getTime() + graceMinutes * 60_000 ? 'present' : 'late';
}

/** Current fine of one attendance record, derived from the ledger. */
export function attendanceBalance(entries: LedgerEntry[], attendanceId: string): number {
  return entries.filter((e) => e.attendanceId === attendanceId).reduce((s, e) => s + e.amount, 0);
}

/** Ledger delta that moves `current` to `target`. Used by adjust, waive and status change. */
export function deltaTo(current: number, target: number): number {
  if (!Number.isInteger(target) || target < 0) throw new RuleError('벌금은 0 이상의 정수여야 합니다');
  return target - current;
}

export function summarize(entries: LedgerEntry[], memberIds: string[]): MemberSummary[] {
  return memberIds.map((userId) => {
    const mine = entries.filter((e) => e.userId === userId);
    const fined = mine.filter((e) => e.kind !== 'payment').reduce((s, e) => s + e.amount, 0);
    const paid = mine.filter((e) => e.kind === 'payment').reduce((s, e) => s + e.amount, 0);
    const outstanding = fined - paid;
    return { userId, fined, paid, outstanding, paidUp: outstanding <= 0 };
  });
}

export function checkPayment(outstanding: number, amount: number): void {
  if (amount > outstanding) throw new RuleError(`미납액(${outstanding}원)보다 많이 납부할 수 없습니다`);
}

import { HttpError } from '../errors.ts';

const hits = new Map<string, number[]>();

export function rateLimit(key: string, limit: number, windowMs: number): void {
  const now = Date.now();
  const recent = (hits.get(key) ?? []).filter((t) => now - t < windowMs);
  if (recent.length >= limit) throw new HttpError(429, '요청이 너무 많습니다. 잠시 후 다시 시도하세요');
  recent.push(now);
  hits.set(key, recent);
}

import { HttpError } from '../errors.ts';
import type { FeePolicy, LedgerEntry, Status } from '../domain/rules.ts';
import { adminClient } from './supabase.ts';

type Res<T> = { data: T | null; error: { code?: string; message: string } | null };

function must<T>(res: Res<T>, conflictMsg = '이미 처리된 요청입니다'): T {
  if (res.error) {
    if (res.error.code === '23505') throw new HttpError(409, conflictMsg);
    if (res.error.code === '40001') throw new HttpError(409, '다른 변경이 먼저 반영되었습니다. 새로고침 후 다시 시도하세요');
    throw new Error(res.error.message);
  }
  if (res.data === null) throw new HttpError(404, '대상을 찾을 수 없습니다');
  return res.data;
}

export async function roleOf(groupId: string, userId: string): Promise<'host' | 'member' | null> {
  const { data, error } = await adminClient().from('group_members').select('role')
    .eq('group_id', groupId).eq('user_id', userId).maybeSingle();
  if (error) throw new Error(error.message);
  return (data?.role as 'host' | 'member' | undefined) ?? null;
}

export async function policyOf(groupId: string): Promise<FeePolicy> {
  const g = must(await adminClient().from('groups').select('late_fee, absent_fee, grace_minutes').eq('id', groupId).maybeSingle());
  return { lateFee: g.late_fee, absentFee: g.absent_fee, graceMinutes: g.grace_minutes };
}

export async function groupByCode(code: string) {
  return must(await adminClient().from('groups').select('id').eq('join_code', code).maybeSingle());
}

export async function getSession(sessionId: string) {
  return must(await adminClient().from('study_sessions').select('id, group_id, starts_at, closed_at').eq('id', sessionId).maybeSingle());
}

export async function getAttendance(attendanceId: string) {
  return must(await adminClient().from('attendance')
    .select('id, user_id, status, study_sessions!inner(group_id)').eq('id', attendanceId).maybeSingle()) as unknown as
    { id: string; user_id: string; status: Status; study_sessions: { group_id: string } };
}

export async function memberIds(groupId: string): Promise<string[]> {
  return must(await adminClient().from('group_members').select('user_id').eq('group_id', groupId)).map((m) => m.user_id);
}

export async function attendedIds(sessionId: string): Promise<string[]> {
  return must(await adminClient().from('attendance').select('user_id').eq('session_id', sessionId)).map((a) => a.user_id);
}

export async function ledger(groupId: string, userId?: string): Promise<LedgerEntry[]> {
  let q = adminClient().from('fine_entries').select('user_id, attendance_id, kind, amount').eq('group_id', groupId);
  if (userId) q = q.eq('user_id', userId);
  return must(await q).map((e) => ({ userId: e.user_id, attendanceId: e.attendance_id, kind: e.kind, amount: e.amount }));
}

export async function rpc(fn: string, args: Record<string, unknown>, conflictMsg?: string): Promise<unknown> {
  const res = await adminClient().rpc(fn, args);
  return must({ data: res.data ?? true, error: res.error }, conflictMsg);
}

export async function insert(table: string, row: Record<string, unknown>): Promise<void> {
  must({ data: true, error: (await adminClient().from(table).insert(row)).error });
}

export async function closeSession(sessionId: string): Promise<boolean> {
  const { data, error } = await adminClient().from('study_sessions').update({ closed_at: new Date().toISOString() })
    .eq('id', sessionId).is('closed_at', null).select('id');
  if (error) throw new Error(error.message);
  return (data ?? []).length > 0;
}

import { ZodError } from 'zod';
import { HttpError } from './errors.ts';
import { RuleError } from './domain/rules.ts';
import { roleOf } from './services/repo.ts';
import { userClient } from './services/supabase.ts';

export function route(fn: (body: unknown, req: Request) => Promise<unknown>) {
  return async (req: Request) => {
    try {
      if (!req.headers.get('content-type')?.includes('application/json')) throw new HttpError(415, 'JSON 요청만 허용됩니다');
      const body = await req.json().catch(() => { throw new HttpError(400, '잘못된 JSON입니다'); });
      return Response.json((await fn(body, req)) ?? { ok: true });
    } catch (e) {
      if (e instanceof HttpError) return Response.json({ error: e.message }, { status: e.status });
      if (e instanceof RuleError) return Response.json({ error: e.message }, { status: 422 });
      if (e instanceof ZodError) return Response.json({ error: e.issues.map((i) => `${i.path.join('.')}: ${i.message}`).join(', ') }, { status: 400 });
      console.error(e);
      return Response.json({ error: '서버 오류가 발생했습니다' }, { status: 500 });
    }
  };
}

export async function requireUser() {
  const { data, error } = await (await userClient()).auth.getUser();
  if (error || !data.user) throw new HttpError(401, '로그인이 만료되었습니다. 다시 로그인하세요');
  return data.user;
}

export async function requireRole(groupId: string, userId: string, role: 'host' | 'member'): Promise<void> {
  const actual = await roleOf(groupId, userId);
  if (!actual) throw new HttpError(403, '모임 멤버가 아닙니다');
  if (role === 'host' && actual !== 'host') throw new HttpError(403, '방장만 할 수 있습니다');
}

export async function middleware(req: NextRequest) {
  let res = NextResponse.next({ request: req });
  const sb = createServerClient(process.env.NEXT_PUBLIC_SUPABASE_URL!, process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!, {
    cookies: {
      getAll: () => req.cookies.getAll(),
      setAll: (list) => {
        list.forEach((c) => req.cookies.set(c.name, c.value));
        res = NextResponse.next({ request: req });
        list.forEach((c) => res.cookies.set(c.name, c.value, c.options));
      },
    },
  });
  const { error } = await sb.auth.getUser(); // refreshes an expiring session
  if (error && error.name !== 'AuthSessionMissingError') console.warn('session refresh failed', error.message);
  return res;
}

import { NextResponse } from 'next/server';
import { HttpError } from '@/lib/errors';
import { rateLimit } from '@/lib/services/rateLimit';
import { env, userClient } from '@/lib/services/supabase';

export async function GET(req: Request) {
  const provider = new URL(req.url).searchParams.get('provider');
  if (provider !== 'kakao' && provider !== 'google') return NextResponse.json({ error: '지원하지 않는 로그인' }, { status: 400 });
  try {
    rateLimit(`login:${req.headers.get('x-forwarded-for') ?? 'unknown'}`, 10, 60_000);
  } catch (e) {
    if (e instanceof HttpError) return NextResponse.json({ error: e.message }, { status: e.status });
    throw e;
  }
  const { data, error } = await (await userClient()).auth.signInWithOAuth({
    provider, options: { redirectTo: `${env('NEXT_PUBLIC_SITE_URL')}/auth/callback` },
  });
  if (error) { console.error(error); return NextResponse.redirect(`${env('NEXT_PUBLIC_SITE_URL')}/?error=login`); }
  return NextResponse.redirect(data.url);
}

import { sessionInput } from '@/lib/domain/inputs';

import { requireRole, requireUser, route } from '@/lib/http';
import { attendedIds, closeSession, getSession, memberIds, policyOf, rpc } from '@/lib/services/repo';

export const POST = route(async (body) => {
  const user = await requireUser();
  const session = await getSession(sessionInput.parse(body).sessionId);
  await requireRole(session.group_id, user.id, 'host');
  // Closing first blocks new check-ins; the unique (session, user) key resolves any in-flight race.
  await closeSession(session.id);
  const [members, attended, policy] = await Promise.all([memberIds(session.group_id), attendedIds(session.id), policyOf(session.group_id)]);
  const absent = members.filter((m) => !attended.includes(m));
  for (const userId of absent) {
    await rpc('record_attendance', { p_session: session.id, p_user: userId, p_status: 'absent', p_charge: fineFor('absent', policy), p_actor: user.id });
  }
  return { absent: absent.length };
});

'use client';
import { useRouter } from 'next/navigation';
import { useState } from 'react';

type Field = { name: string; label: string; type?: 'text' | 'number' | 'datetime-local'; options?: string[] };

export function ApiForm({ action, label, fields = [], hidden = {} }:
  { action: string; label: string; fields?: Field[]; hidden?: Record<string, string> }) {
  const router = useRouter();
  const [error, setError] = useState('');
  const [busy, setBusy] = useState(false);

  async function submit(e: React.FormEvent<HTMLFormElement>) {
    e.preventDefault();
    setBusy(true); setError('');
    const body: Record<string, unknown> = { ...hidden };
    new FormData(e.currentTarget).forEach((v, k) => {
      const f = fields.find((x) => x.name === k);
      if (v === '') return;
      body[k] = f?.type === 'datetime-local' ? new Date(String(v)).toISOString() : v;
    });
    try {
      const res = await fetch(action, { method: 'POST', headers: { 'content-type': 'application/json' }, body: JSON.stringify(body) });
      if (res.status === 401) { router.push('/?expired=1'); return; }
      const json = await res.json();
      if (!res.ok) { setError(json.error ?? `요청 실패 (${res.status})`); return; }
      router.refresh();
    } catch (err) {
      setError(err instanceof Error ? err.message : '네트워크 오류');
    } finally { setBusy(false); }
  }

  return (
    <form onSubmit={submit} className="flex flex-wrap items-end gap-2">
      {fields.map((f) => (
        <label key={f.name} className="flex flex-col text-xs text-zinc-600">{f.label}
          {f.options
            ? <select name={f.name} className="rounded border px-2 py-1 text-sm">{f.options.map((o) => <option key={o}>{o}</option>)}</select>
            : <input name={f.name} type={f.type ?? 'text'} className="rounded border px-2 py-1 text-sm" />}
        </label>
      ))}
      <button disabled={busy} className="rounded bg-zinc-900 px-3 py-1 text-sm text-white disabled:opacity-50">{label}</button>
      {error && <p role="alert" className="w-full text-sm text-red-600">{error}</p>}
    </form>
  );
}

  const { id } = await params;
  const sb = await userClient();
  const { data: { user } } = await sb.auth.getUser();
  if (!user) redirect('/?expired=1');
  // RLS limits every read below to groups the user belongs to.
  const [g, m, s, f] = await Promise.all([
    sb.from('groups').select('name, join_code, late_fee, absent_fee, grace_minutes').eq('id', id).maybeSingle(),
    sb.from('group_members').select('user_id, role, profiles(display_name)').eq('group_id', id),
    sb.from('study_sessions').select('id, starts_at, closed_at, attendance(id, user_id, status)').eq('group_id', id).order('starts_at', { ascending: false }).limit(20),
    sb.from('fine_entries').select('user_id, attendance_id, kind, amount').eq('group_id', id),
  ]);
  for (const r of [g, m, s, f]) if (r.error) throw new Error(r.error.message);
  if (!g.data) notFound();
  const members = (m.data ?? []) as unknown as { user_id: string; role: string; profiles: { display_name: string } }[];
  const sessions = (s.data ?? []) as { id: string; starts_at: string; closed_at: string | null; attendance: { id: string; user_id: string; status: keyof typeof STATUS }[] }[];
