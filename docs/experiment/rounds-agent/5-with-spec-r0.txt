=== docs/decisions.md ===
# Design decisions

1. **Delayed is a derived flag, not a fourth status.** Reason: the owner fixed status to exactly 3 values; delay is computed from `due_date` + `status` on every read, so it can never go stale. Rejected: storing `delayed` in the DB or a nightly job that rewrites status.
2. **"Today" is computed on the server in one time zone (`APP_TIMEZONE`, default Asia/Seoul), and `due_date` is a `date` column.** Reason: a due date is a calendar day for the team, so comparing `YYYY-MM-DD` strings in one fixed zone avoids browser/server time-zone drift. Rejected: `timestamptz` deadlines compared against the browser clock.
3. **Owner-only rule is enforced three times: domain check in API routes, RLS policies, and a DB trigger on `tasks`.** Reason: members may change task status, but only the owner may change title, due date or role; RLS alone cannot restrict by column, so a trigger guards the plan fields. Rejected: hiding buttons in the UI only.
4. **API routes use the user's own Supabase session (anon key + cookies), never a service-role key.** Reason: RLS then applies to every query, so a bug in a route cannot leak another project. Rejected: service-role client with manual filtering.
5. **Members are added by email through the `add_member` SECURITY DEFINER function.** Reason: profiles (emails) are only visible to co-members; the function checks ownership and looks the email up without exposing the profiles table. Rejected: a public profiles select policy.
6. **Composite foreign keys keep roles and assignees inside the same project.** `tasks(role_id, project_id) -> roles(id, project_id)` and `roles(project_id, assignee_id) -> project_members`. Reason: the DB itself rejects cross-project references. Rejected: checking only in application code.
7. **Login/signup rate limit is in-memory per server instance.** Reason: no extra paid service; Supabase Auth also applies its own limits. Known limit: counters reset on cold start and are not shared across instances. Rejected: Redis (cost).

export const TASK_STATUSES = ["todo", "in_progress", "done"] as const;
export type TaskStatus = (typeof TASK_STATUSES)[number];
export const STATUS_LABEL: Record<TaskStatus, string> = { todo: "할 일", in_progress: "진행 중", done: "완료" };
export const DELAYED_LABEL = "지연";

/** Calendar day (YYYY-MM-DD) in the given IANA time zone. */
export function todayIn(timeZone: string, now: Date = new Date()): string {
  return new Intl.DateTimeFormat("en-CA", { timeZone, year: "numeric", month: "2-digit", day: "2-digit" }).format(now);
}

/** Rule: an unfinished task whose due date has passed is delayed. A task due today is not delayed yet. */
export function isDelayed(task: { status: TaskStatus; due_date: string | null }, today: string): boolean {
  return task.status !== "done" && task.due_date !== null && task.due_date < today;
}

export function withDelay<T extends { status: TaskStatus; due_date: string | null }>(tasks: T[], today: string) {
  return tasks.map((t) => ({ ...t, delayed: isDelayed(t, today) }));
}

import { DomainError } from "./errors";
import { TASK_STATUSES, type TaskStatus } from "./task";

const UUID = /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i;
const EMAIL = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
const invalid = (m: string) => new DomainError("invalid", m);

export function asObject(v: unknown): Record<string, unknown> {
  if (typeof v !== "object" || v === null || Array.isArray(v)) throw invalid("요청 본문이 올바르지 않습니다.");
  return v as Record<string, unknown>;
}
export function text(v: unknown, field: string, max: number): string {
  if (typeof v !== "string") throw invalid(`${field}: 문자열이어야 합니다.`);
  const t = v.trim();
  if (!t || t.length > max) throw invalid(`${field}: 1~${max}자여야 합니다.`);
  return t;
}
export function email(v: unknown): string {
  const e = text(v, "이메일", 254).toLowerCase();
  if (!EMAIL.test(e)) throw invalid("이메일 형식이 올바르지 않습니다.");
  return e;
}
export function password(v: unknown): string {
  if (typeof v !== "string" || v.length < 8 || v.length > 72) throw invalid("비밀번호는 8~72자여야 합니다.");
  return v;
}
/** Path ids: a malformed id is treated as "not found". */
export function pathId(v: unknown): string {
  if (typeof v !== "string" || !UUID.test(v)) throw new DomainError("not_found", "대상을 찾을 수 없습니다.");
  return v;
}
export function optionalId(v: unknown, field: string): string | null {
  if (v === null || v === undefined || v === "") return null;
  if (typeof v !== "string" || !UUID.test(v)) throw invalid(`${field}: 올바른 ID가 아닙니다.`);
  return v;
}
export function optionalDate(v: unknown): string | null {
  if (v === null || v === undefined || v === "") return null;
  if (typeof v !== "string" || !/^\d{4}-\d{2}-\d{2}$/.test(v)) throw invalid("마감일은 YYYY-MM-DD 형식이어야 합니다.");
  const d = new Date(`${v}T00:00:00Z`);
  if (Number.isNaN(d.getTime()) || d.toISOString().slice(0, 10) !== v) throw invalid("존재하지 않는 날짜입니다.");
  return v;
}
export function status(v: unknown): TaskStatus {
  if (!TASK_STATUSES.includes(v as TaskStatus)) throw invalid("상태는 할 일, 진행 중, 완료 중 하나여야 합니다.");
  return v as TaskStatus;
}

export interface TaskPatch { title?: string; status?: TaskStatus; due_date?: string | null; role_id?: string | null }

export function parseTaskCreate(raw: unknown) {
  const b = asObject(raw);
  return { title: text(b.title, "작업명", 200), due_date: optionalDate(b.due_date), role_id: optionalId(b.role_id, "역할") };
}
export function parseTaskPatch(raw: unknown): TaskPatch {
  const b = asObject(raw);
  const patch: TaskPatch = {};
  if ("title" in b) patch.title = text(b.title, "작업명", 200);
  if ("status" in b) patch.status = status(b.status);
  if ("due_date" in b) patch.due_date = optionalDate(b.due_date);
  if ("role_id" in b) patch.role_id = optionalId(b.role_id, "역할");
  if (Object.keys(patch).length === 0) throw invalid("변경할 항목이 없습니다.");
  return patch;
}

import { DomainError } from "../domain/errors";

// In-memory, per instance (see docs/decisions.md #7).
const hits = new Map<string, number[]>();

export function rateLimit(key: string, limit: number, windowMs: number): void {
  const now = Date.now();
  const recent = (hits.get(key) ?? []).filter((t) => now - t < windowMs);
  if (recent.length >= limit) throw new DomainError("rate_limited", "요청이 너무 많습니다. 잠시 후 다시 시도하세요.");
  recent.push(now);
  hits.set(key, recent);
}

import { DomainError, type ErrorKind } from "../domain/errors";
import { asObject } from "../domain/validate";

const STATUS: Record<ErrorKind, number> = { invalid: 400, unauthorized: 401, forbidden: 403, not_found: 404, rate_limited: 429 };

export async function respond(fn: () => Promise<unknown>): Promise<NextResponse> {
  try {
    return NextResponse.json((await fn()) ?? { ok: true });
  } catch (e) {
    if (e instanceof DomainError) return NextResponse.json({ error: e.message }, { status: STATUS[e.kind] });
    console.error(e);
    return NextResponse.json({ error: "서버 오류가 발생했습니다." }, { status: 500 });
  }
}
export async function jsonBody(req: Request): Promise<Record<string, unknown>> {

    throw new DomainError("invalid", `JSON 본문이 필요합니다. (${e instanceof Error ? e.message : String(e)})`);
  });
  return asObject(raw);
}

// Refreshes the auth cookie on every request so login survives page reloads.
export async function middleware(req: NextRequest) {
  let res = NextResponse.next({ request: req });
  const sb = createServerClient(env("NEXT_PUBLIC_SUPABASE_URL"), env("NEXT_PUBLIC_SUPABASE_ANON_KEY"), {

        res = NextResponse.next({ request: req });

      },
    },
  });
  await sb.auth.getUser();
  return res;

import { DomainError } from "@/lib/domain/errors";
import { email, password } from "@/lib/domain/validate";
import { clientIp, jsonBody, respond } from "@/lib/server/http";
import { rateLimit } from "@/lib/services/rateLimit";
import { supabaseForRequest } from "@/lib/services/supabase";

export async function POST(req: Request, { params }: { params: Promise<{ action: string }> }) {
  return respond(async () => {
    const { action } = await params;
    const sb = await supabaseForRequest();
    if (action === "logout") {
      const { error } = await sb.auth.signOut();
      if (error) throw error;
      return { ok: true };
    }
    if (action !== "login" && action !== "signup") throw new DomainError("not_found", "알 수 없는 요청입니다.");
    rateLimit(`${action}:${clientIp(req)}`, action === "login" ? 10 : 5, 60_000);
    const b = await jsonBody(req);
    const creds = { email: email(b.email), password: password(b.password) };
    if (action === "login") {
      rateLimit(`login-email:${creds.email}`, 10, 10 * 60_000);
      const { error } = await sb.auth.signInWithPassword(creds);
      if (error) throw new DomainError("unauthorized", "이메일 또는 비밀번호가 올바르지 않습니다.");
      return { ok: true };
    }
    const { data, error } = await sb.auth.signUp(creds);
    if (error) throw new DomainError("invalid", `회원가입에 실패했습니다: ${error.message}`);
    return { ok: true, needsConfirmation: !data.session };
  });
}

import { jsonBody, requireUser, respond } from "@/lib/server/http";

// POST /api/projects/:id/{members|roles|tasks} — all are owner-only plan changes.
export async function POST(req: Request, { params }: { params: Promise<{ id: string; kind: string }> }) {
  return respond(async () => {
    const { id: rawId, kind } = await params;
    const id = pathId(rawId);
    const sb = await supabaseForRequest();
    const user = await requireUser(sb);
    const project = await getProject(sb, id);
    assertOwner(project.owner_id, user.id);
    const b = await jsonBody(req);
    if (kind === "members") return addMember(sb, id, email(b.email));
    if (kind === "roles") return { role: await createRole(sb, id, text(b.name, "역할명", 50), optionalId(b.assignee_id, "담당자")) };
    if (kind === "tasks") return { task: await createTask(sb, id, parseTaskCreate(b)) };
    throw new DomainError("not_found", "알 수 없는 요청입니다.");
  });
}

  return respond(async () => {
    const id = pathId((await params).id);
    const sb = await supabaseForRequest();

export async function api<T = unknown>(path: string, method = "GET", body?: unknown): Promise<T> {
  const res = await fetch(path, {
    method,
    headers: body === undefined ? undefined : { "Content-Type": "application/json" },
    body: body === undefined ? undefined : JSON.stringify(body),
  });
  const data = await res.json().catch(() => ({ error: `응답을 읽을 수 없습니다 (${res.status})` }));
  if (!res.ok) throw new Error(data.error ?? `요청 실패 (${res.status})`);
  return data as T;
}

import { useState } from "react";

type NoticeState = { ok: boolean; text: string } | null;

/** Runs an action and shows its success or failure; failures are never silent. */
export function useNotice() {
  const [notice, setNotice] = useState<NoticeState>(null);
  async function run(fn: () => Promise<unknown>, okText?: string): Promise<boolean> {
    try {
      await fn();
      setNotice(okText ? { ok: true, text: okText } : null);
      return true;
    } catch (e) {
      setNotice({ ok: false, text: e instanceof Error ? e.message : String(e) });
      return false;
    }
  }
  return { notice, run };
}

export function Notice({ notice }: { notice: NoticeState }) {
  if (!notice) return null;
  return <p role="status" className={`my-2 text-sm ${notice.ok ? "text-green-700" : "text-red-700"}`}>{notice.text}</p>;
}
