// 메모리 기반 고정 창(fixed window) 요청 제한. 서버 1대 기준이며, 재시작하면 초기화됩니다.
// 여러 대로 늘리면 Redis 같은 공유 저장소로 옮겨야 합니다.
export function rateLimit({ windowMs, max, key, message }) {
  const hits = new Map();
  setInterval(() => {
    const now = Date.now();
    for (const [k, v] of hits) if (v.resetAt <= now) hits.delete(k);
  }, windowMs).unref();

  return (req, res, next) => {
    const k = key(req);
    const now = Date.now();
    let entry = hits.get(k);
    if (!entry || entry.resetAt <= now) {
      entry = { count: 0, resetAt: now + windowMs };
      hits.set(k, entry);
    }
    entry.count += 1;
    if (entry.count > max) {
      res.set('Retry-After', String(Math.ceil((entry.resetAt - now) / 1000)));
      return res.status(429).json({ error: message });
    }
    next();
  };
}

app.post('/api/auth/request-code', codeIpLimit, async (req, res) => {
  const email = str(req.body.email, 200).toLowerCase();
  if (!isSchoolEmail(email)) return res.status(400).json({ error: `@${DOMAIN} 이메일만 가입할 수 있습니다.` });

  const prev = db.prepare('SELECT expires_at FROM verifications WHERE email = ?').get(email);
  if (prev && prev.expires_at - 9 * 60e3 > Date.now()) {
    return res.status(429).json({ error: '1분 후 다시 요청해 주세요.' });
  }
  const code = String(crypto.randomInt(100000, 1000000));
  db.prepare(`INSERT INTO verifications (email, code, expires_at, attempts) VALUES (?, ?, ?, 0)
    ON CONFLICT(email) DO UPDATE SET code = excluded.code, expires_at = excluded.expires_at, attempts = 0`)
    .run(email, code, Date.now() + 10 * 60e3);
  await sendVerificationCode(email, code);
  res.json({ ok: true });

app.post('/api/auth/signup', signupIpLimit, async (req, res) => {
  const email = str(req.body.email, 200).toLowerCase();
  const code = str(req.body.code, 10);
  const nickname = str(req.body.nickname, 20);
  const password = typeof req.body.password === 'string' ? req.body.password : '';
  if (!isSchoolEmail(email)) return res.status(400).json({ error: `@${DOMAIN} 이메일만 가입할 수 있습니다.` });
  if (!nickname) return res.status(400).json({ error: '닉네임을 입력하세요.' });
  if (password.length < 8) return res.status(400).json({ error: '비밀번호는 8자 이상이어야 합니다.' });

  const v = db.prepare('SELECT * FROM verifications WHERE email = ?').get(email);
  if (!v || v.expires_at < Date.now()) return res.status(400).json({ error: '인증 코드를 다시 요청하세요.' });
  if (v.attempts >= 5) return res.status(429).json({ error: '시도 횟수를 넘었습니다. 코드를 다시 요청하세요.' });
  if (v.code !== code) {
    db.prepare('UPDATE verifications SET attempts = attempts + 1 WHERE email = ?').run(email);
    return res.status(400).json({ error: '인증 코드가 틀렸습니다.' });
  }
  const existing = db.prepare('SELECT verified FROM users WHERE email = ?').get(email);
  if (existing?.verified) return res.status(409).json({ error: '이미 가입된 이메일입니다.' });

  const hash = await bcrypt.hash(password, 10);
  db.prepare(`INSERT INTO users (email, nickname, password_hash, verified) VALUES (?, ?, ?, 1)
    ON CONFLICT(email) DO UPDATE SET nickname = excluded.nickname, password_hash = excluded.password_hash, verified = 1`)
    .run(email, nickname, hash);
  db.prepare('DELETE FROM verifications WHERE email = ?').run(email);
  const user = db.prepare('SELECT id FROM users WHERE email = ?').get(email);
  req.session.regenerate(() => {
    req.session.userId = user.id;
    res.json({ ok: true });
  });
});

app.post('/api/auth/login', loginIpLimit, loginEmailLimit, async (req, res) => {
  const email = str(req.body.email, 200).toLowerCase();

  }
  req.session.regenerate(() => {
    req.session.userId = user.id;
    res.json({ ok: true });
  });
});

app.get('/api/posts/:id', requireLogin, (req, res) => {
  const post = db.prepare(`SELECT p.*, u.nickname, u.email FROM posts p JOIN users u ON u.id = p.user_id WHERE p.id = ?`)
    .get(Number(req.params.id));
  if (!post) return res.status(404).json({ error: '글이 없습니다.' });
  const comments = db.prepare(`SELECT c.id, c.body, c.created_at, c.user_id, u.nickname FROM comments c
    JOIN users u ON u.id = c.user_id WHERE c.post_id = ? ORDER BY c.id`).all(post.id);
  res.json({ post: { ...post, mine: post.user_id === req.session.userId }, comments });
});

function readPost(body) {
  const type = body.type === 'buy' ? 'buy' : 'sell';
  const title = str(body.title, 100);
  const price = Number(body.price);
  if (!title) return { error: '교재명을 입력하세요.' };
  if (!Number.isInteger(price) || price < 0 || price > 10_000_000) return { error: '가격을 확인하세요.' };
  return { type, title, price, course: str(body.course, 100), condition: str(body.condition, 30), body: str(body.body, 2000) };
}

app.post('/api/posts', requireLogin, (req, res) => {
  const p = readPost(req.body);
  if (p.error) return res.status(400).json(p);
  const r = db.prepare(`INSERT INTO posts (user_id, type, title, course, price, condition, body) VALUES (?, ?, ?, ?, ?, ?, ?)`)
    .run(req.session.userId, p.type, p.title, p.course, p.price, p.condition, p.body);
  res.json({ id: r.lastInsertRowid });
});

function ownPost(req, res) {
  const post = db.prepare('SELECT * FROM posts WHERE id = ?').get(Number(req.params.id));
  if (!post) { res.status(404).json({ error: '글이 없습니다.' }); return null; }
  if (post.user_id !== req.session.userId) { res.status(403).json({ error: '본인 글만 수정할 수 있습니다.' }); return null; }
  return post;
}

app.put('/api/posts/:id', requireLogin, (req, res) => {
  const post = ownPost(req, res);
  if (!post) return;
  const p = readPost(req.body);
  if (p.error) return res.status(400).json(p);
  const status = ['open', 'reserved', 'done'].includes(req.body.status) ? req.body.status : post.status;
  db.prepare(`UPDATE posts SET type=?, title=?, course=?, price=?, condition=?, body=?, status=? WHERE id=?`)
    .run(p.type, p.title, p.course, p.price, p.condition, p.body, status, post.id);
  res.json({ ok: true });
});

app.delete('/api/posts/:id', requireLogin, (req, res) => {
  const post = ownPost(req, res);
  if (!post) return;
  db.prepare('DELETE FROM posts WHERE id = ?').run(post.id);
  res.json({ ok: true });
});

app.post('/api/posts/:id/comments', requireLogin, (req, res) => {
  const body = str(req.body.body, 500);
  if (!body) return res.status(400).json({ error: '내용을 입력하세요.' });

  if (!post) return res.status(404).json({ error: '글이 없습니다.' });
  db.prepare('INSERT INTO comments (post_id, user_id, body) VALUES (?, ?, ?)').run(post.id, req.session.userId, body);
  res.json({ ok: true });
});

async function api(path, opts = {}) {
  $err.textContent = '';
  const res = await fetch(path, { headers: { 'Content-Type': 'application/json' }, ...opts, body: opts.body && JSON.stringify(opts.body) });
  const data = await res.json().catch(() => ({}));
  if (!res.ok) { $err.textContent = data.error || '오류가 발생했습니다.'; throw new Error(data.error); }
  return data;
}

# 설계 결정 기록

## 1. 가입 자격: 학교 이메일 도메인 + 인증 코드
- 결정: `SCHOOL_DOMAIN`(예: myschool.ac.kr)과 정확히 일치하는 도메인만 받고, 6자리 코드(10분 유효, 5회 오입력 시 무효)로 메일 소유를 확인한 뒤 계정을 활성화한다.
- 이유: 도메인 문자열만 보면 남의 학교 메일로도 가입할 수 있다. 메일 소유 확인이 있어야 "우리 학교 학생끼리"가 성립한다.
- 한계: 졸업·휴학생 메일이 살아 있으면 계속 이용할 수 있다. 하위 도메인(student.myschool.ac.kr 등)은 지금 받지 않는다.

## 2. 메일 발송은 콘솔 출력으로 대체
- 결정: `mailer.js`가 코드를 서버 콘솔에 출력한다.
- 이유: SMTP 계정 없이 바로 실행해 볼 수 있게. 운영 전에 반드시 실제 발송으로 교체해야 한다.

## 3. 게시판은 로그인한 학생만 열람
- 결정: 목록·상세·댓글 모두 로그인 필요.
- 이유: 판매자 닉네임·거래 장소 등 학내 정보가 외부에 노출되지 않게 한다.

## 4. 거래는 앱 밖에서, 문의는 댓글로
- 결정: 결제·채팅 기능 없이 글 + 댓글 + 상태(판매중/예약중/거래완료)만 둔다.
- 이유: 학내 직거래가 대부분이라 결제 대행은 범위 밖. 결제를 넣으면 분쟁·환불 책임이 생긴다.

## 5. 저장소: SQLite 단일 파일
- 결정: better-sqlite3, `market.db` 한 파일.
- 이유: 한 학교 규모(수천 명)에서는 충분하고 설치·백업이 간단하다. 서버를 여러 대로 늘릴 때 Postgres 등으로 옮긴다.

## 6. 인증 방식: 서버 세션 쿠키
- 결정: express-session, httpOnly + sameSite=lax 쿠키, 로그인·가입 시 세션 재생성. 비밀번호는 bcrypt(10).
- 이유: 단일 서버 웹앱에서 JWT보다 단순하고 즉시 로그아웃이 된다.
- 한계: 기본 메모리 세션 저장소라 재시작하면 모두 로그아웃된다. 운영에서는 세션 저장소를 따로 둔다.

## 7. 요청 횟수 제한 (rate limiting)
- 결정: `rateLimit.js`의 메모리 고정 창 방식.
  - 로그인: IP당 15분 20회 + 이메일당 15분 10회
  - 인증 코드 요청: IP당 1시간 10회 (이메일당 1분 1회 제한은 기존 DB 검사로 유지)
  - 가입(코드 확인): IP당 15분 20회 (코드별 5회 오입력 제한은 기존대로 유지)
- 이유: 비밀번호 대입, 인증 코드 대입, 메일 폭탄을 막는다. IP와 이메일 두 기준을 같이 써서 한 IP의 여러 계정 공격과 여러 IP의 한 계정 공격을 모두 늦춘다.
- 한계: 서버 메모리라 재시작 시 초기화되고 여러 대에서는 공유되지 않는다. 프록시 뒤에서는 `app.set('trust proxy', ...)`를 맞게 설정해야 `req.ip`가 실제 IP가 된다. 이메일 기준 제한은 공격자가 피해자 로그인을 15분간 막는 데 쓸 수 있다.

## 8. 보안 기본값
- 사용자 입력은 화면에 넣기 전 HTML 이스케이프, SQL은 모두 바인딩 파라미터.
- 글 수정·삭제는 작성자 본인만.

# 교재 장터
우리 학교 학생끼리 중고 교재를 사고파는 게시판입니다.

- 실행: `npm install` 후 `SCHOOL_DOMAIN=myschool.ac.kr SESSION_SECRET=... npm start`
- 가입: 학교 이메일(@SCHOOL_DOMAIN)로 인증 코드를 받아 가입합니다. 지금은 코드가 서버 콘솔에 출력되므로 `mailer.js`를 SMTP 발송으로 바꿔야 실제 운영할 수 있습니다.
- 기능: 팝니다/삽니다 글쓰기, 검색, 상태(판매중/예약중/거래완료), 댓글로 거래 문의. 로그인한 학생만 글을 볼 수 있습니다.
- 설계 결정은 `docs/decisions.md`에 있습니다.
